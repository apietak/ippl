16 template <typename T, unsigned Dim, unsigned numElementDOFs>
17 struct EvalFunctor {
18 const Vector<T, Dim> DPhiInvT;
19 const T absDetDPhi;
20
21 EvalFunctor(Vector<T, Dim> DPhiInvT, T absDetDPhi)
22 : DPhiInvT(DPhiInvT)
23 , absDetDPhi(absDetDPhi) {}
24
25 KOKKOS_FUNCTION const auto operator()(size_t i, size_t j,
26 const ippl::Vector<ippl::Vector<T, Dim>, numElementDOFs>& curl_b_q_k,
27 const ippl::Vector<ippl::Vector<T, Dim>, numElementDOFs>& val_b_q_k, bool onBoundary) const {
28
29 T curlTerm = dot(DPhiInvT*curl_b_q_k[j], DPhiInvT*curl_b_q_k[i]).apply();
30 T massTerm = dot(val_b_q_k[j], val_b_q_k[i]).apply();
31 return (curlTerm + massTerm)*absDetDPhi;
32 }
33 };

130 // start a timer
131 static IpplTimings::TimerRef solve = IpplTimings::getTimer("solve");
132 IpplTimings::startTimer(solve);
133
134 const Vector<size_t, Dim> zeroNdIndex = Vector<size_t, Dim>(0);
135
136 // We can pass the zeroNdIndex here, since the transformation
137 // jacobian does not depend on translation
138 const auto firstElementVertexPoints =
139 nedelecSpace_m.getElementMeshVertexPoints(zeroNdIndex);
140
141 // Compute Inverse Transpose Transformation Jacobian ()
142 const Vector<T, Dim> DPhiInvT =
143 refElement_m.getInverseTransposeTransformationJacobian(firstElementVertexPoints);
144
145 // Compute absolute value of the determinant of the transformation
146 // jacobian (|det D Phi_K|)
147 const T absDetDPhi = Kokkos::abs(
148 refElement_m.getDeterminantOfTransformationJacobian(firstElementVertexPoints));

150 EvalFunctor<T, Dim, this->nedelecSpace_m.numElementDOFs> maxwellDiffusionEval(
151 DPhiInvT, absDetDPhi);

159 vector.fillHalo();

161 FEMVector<T> return_vector = nedelecSpace_m.evaluateAx(vector,maxwellDiffusionEval);
162
163 return_vector.accumulateHalo();
164
165 IpplTimings::stopTimer(opTimer);
