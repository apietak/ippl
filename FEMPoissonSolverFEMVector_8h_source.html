16 template <typename Tlhs, unsigned Dim, unsigned numElemDOFs>
17 struct EvalFunctor {
18 const Vector<Tlhs, Dim> DPhiInvT;
19 const Tlhs absDetDPhi;
20
21 EvalFunctor(Vector<Tlhs, Dim> DPhiInvT, Tlhs absDetDPhi)
22 : DPhiInvT(DPhiInvT)
23 , absDetDPhi(absDetDPhi) {}
24
25 KOKKOS_FUNCTION const auto operator()(
26 const size_t& i, const size_t& j,
27 const Vector<Vector<Tlhs, Dim>, numElemDOFs>& grad_b_q_k) const {
28 return dot((DPhiInvT * grad_b_q_k[j]), (DPhiInvT * grad_b_q_k[i])).apply() * absDetDPhi;
29 }
30 };

133 void solve() override {
134 // start a timer
135 static IpplTimings::TimerRef solve = IpplTimings::getTimer("solve");
136 IpplTimings::startTimer(solve);
137
138 const Vector<size_t, Dim> zeroNdIndex = Vector<size_t, Dim>(0);
139
140 // We can pass the zeroNdIndex here, since the transformation
141 // jacobian does not depend on translation
142 const auto firstElementVertexPoints =
143 lagrangeSpace_m.getElementMeshVertexPoints(zeroNdIndex);
144
145 // Compute Inverse Transpose Transformation Jacobian ()
146 const Vector<Tlhs, Dim> DPhiInvT =
147 refElement_m.getInverseTransposeTransformationJacobian(firstElementVertexPoints);
148
149 // Compute absolute value of the determinant of the transformation
150 // jacobian (|det D Phi_K|)
151 const Tlhs absDetDPhi = Kokkos::abs(
152 refElement_m.getDeterminantOfTransformationJacobian(firstElementVertexPoints));
153
154 EvalFunctor<Tlhs, Dim, this->lagrangeSpace_m.numElementDOFs> poissonEquationEval(
155 DPhiInvT, absDetDPhi);
156
157 const auto algoOperator = [poissonEquationEval, this](FEMVector<Tlhs> vector)
158 -> FEMVector<Tlhs> {
159 // start a timer
160 static IpplTimings::TimerRef opTimer = IpplTimings::getTimer("operator");
161 IpplTimings::startTimer(opTimer);
162
163 vector.fillHalo();
164
165 FEMVector<Tlhs> return_vector = lagrangeSpace_m.evaluateAx(vector,
166 poissonEquationEval);
167
168 return_vector.accumulateHalo();
169
170 IpplTimings::stopTimer(opTimer);
171
172 return return_vector;
173 };
174
175 pcg_algo_m.setOperator(algoOperator);
176
177
178
179
180
181 // start a timer
182 static IpplTimings::TimerRef pcgTimer = IpplTimings::getTimer("pcg");
183 IpplTimings::startTimer(pcgTimer);
184
185 FEMVector<Tlhs> lhsVector = lagrangeSpace_m.interpolateToFEMVector(*(this->lhs_mp));
186
187 try {
188 pcg_algo_m(lhsVector, *rhsVector_m, this->params_m);
189 } catch (IpplException& e) {

191 }

193 lhsVector.fillHalo();
194
195 IpplTimings::stopTimer(pcgTimer);
196
197
198 lagrangeSpace_m.reconstructToField(lhsVector, *(this->lhs_mp));
199 lagrangeSpace_m.reconstructToField(*rhsVector_m, *(this->rhs_mp));
200
201
202 IpplTimings::stopTimer(solve);
203 }
