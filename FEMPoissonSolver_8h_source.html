15 template <typename Tlhs, unsigned Dim, unsigned numElemDOFs>
16 struct EvalFunctor {
17 const Vector<Tlhs, Dim> DPhiInvT;
18 const Tlhs absDetDPhi;
19
20 EvalFunctor(Vector<Tlhs, Dim> DPhiInvT, Tlhs absDetDPhi)
21 : DPhiInvT(DPhiInvT)
22 , absDetDPhi(absDetDPhi) {}
23
24 KOKKOS_FUNCTION const auto operator()(
25 const size_t& i, const size_t& j,
26 const Vector<Vector<Tlhs, Dim>, numElemDOFs>& grad_b_q_k) const {
27 return dot((DPhiInvT * grad_b_q_k[j]), (DPhiInvT * grad_b_q_k[i])).apply() * absDetDPhi;
28 }
29 };

110 void solve() override {
111 // start a timer
112 static IpplTimings::TimerRef solve = IpplTimings::getTimer("solve");
113 IpplTimings::startTimer(solve);
114
115 const Vector<size_t, Dim> zeroNdIndex = Vector<size_t, Dim>(0);
116
117 // We can pass the zeroNdIndex here, since the transformation jacobian does not depend
118 // on translation
119 const auto firstElementVertexPoints =
120 lagrangeSpace_m.getElementMeshVertexPoints(zeroNdIndex);
121
122 // Compute Inverse Transpose Transformation Jacobian ()
123 const Vector<Tlhs, Dim> DPhiInvT =
124 refElement_m.getInverseTransposeTransformationJacobian(firstElementVertexPoints);
125
126 // Compute absolute value of the determinant of the transformation jacobian (|det D
127 // Phi_K|)
128 const Tlhs absDetDPhi = Kokkos::abs(
129 refElement_m.getDeterminantOfTransformationJacobian(firstElementVertexPoints));
130
131 EvalFunctor<Tlhs, Dim, this->lagrangeSpace_m.numElementDOFs> poissonEquationEval(
132 DPhiInvT, absDetDPhi);
133
134 const auto algoOperator = [poissonEquationEval, this](lhs_type field) -> lhs_type {
135 // start a timer
136 static IpplTimings::TimerRef opTimer = IpplTimings::getTimer("operator");
137 IpplTimings::startTimer(opTimer);
138
139 field.fillHalo();
140
141 auto return_field = lagrangeSpace_m.evaluateAx(field, poissonEquationEval);
142
143 return_field.accumulateHalo();
144
145 IpplTimings::stopTimer(opTimer);
146
147 return return_field;
148 };
149
150 pcg_algo_m.setOperator(algoOperator);
151
152 // start a timer
153 static IpplTimings::TimerRef pcgTimer = IpplTimings::getTimer("pcg");
154 IpplTimings::startTimer(pcgTimer);
155
156 pcg_algo_m(*(this->lhs_mp), *(this->rhs_mp), this->params_m);
157
158 (this->lhs_mp)->fillHalo();
159
160 IpplTimings::stopTimer(pcgTimer);
161
162 int output = this->params_m.template get<int>("output_type");
163 if (output & Base::GRAD) {
164 *(this->grad_mp) = -grad(*(this->lhs_mp));
165 }
166
167 IpplTimings::stopTimer(solve);
168 }
