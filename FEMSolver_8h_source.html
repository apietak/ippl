371 KOKKOS_FUNCTION size_t getElementIndex(const indices_t& ndindex) const {
372 size_t element_index = 0;
373
374 // This is the number of cells in each dimension. It is one less than the number of
375 // vertices in each dimension, which is returned by Mesh::getGridsize().
376 ippl::Vector<size_t, Dim> cells_per_dim = space_m.nr_m - 1;
377
378 size_t remaining_number_of_cells = 1;
379
380 for (unsigned int d = 0; d < Dim; ++d) {
381 element_index += ndindex[d] * remaining_number_of_cells;
382 remaining_number_of_cells *= cells_per_dim[d];
383 }
384
385 return element_index;
386 }

389 void initializeElementIndices(const Layout_t& layout) {
390 const auto& ldom = layout.getLocalNDIndex();
391 int npoints = ldom.size();
392 auto first = ldom.first();
393 auto last = ldom.last();
394 ippl::Vector<double, Dim> bounds;
395
396 for (size_t d = 0; d < Dim; ++d) {
397 bounds[d] = space_m.nr_m[d] - 1;
398 }
399
400 int upperBoundaryPoints = -1;
401
402 Kokkos::View<size_t*> points("ComputeMapping", npoints);
403 Kokkos::parallel_reduce(
404 "ComputePoints", npoints,
405 KOKKOS_CLASS_LAMBDA(const int i, int& local) {
406 int idx = i;
407 indices_t val;
408 bool isBoundary = false;
409 for (unsigned int d = 0; d < Dim; ++d) {
410 int range = last[d] - first[d] + 1;
411 val[d] = first[d] + (idx % range);
412 idx /= range;
413 if (val[d] == bounds[d]) {
414 isBoundary = true;
415 }
416 }
417 points(i) = (!isBoundary) * (this->getElementIndex(val));
418 local += isBoundary;
419 },
420 Kokkos::Sum<int>(upperBoundaryPoints));
421 Kokkos::fence();
422
423 int elementsPerRank = npoints - upperBoundaryPoints;
424 elementIndices = Kokkos::View<size_t*>("i", elementsPerRank);
425 Kokkos::View<size_t> index("index");
426
427 Kokkos::parallel_for(
428 "RemoveNaNs", npoints, KOKKOS_CLASS_LAMBDA(const int i) {
429 if ((points(i) != 0) || (i == 0)) {
430 const size_t idx = Kokkos::atomic_fetch_add(&index(), 1);
431 elementIndices(idx) = points(i);
432 }
433 });
434
435 for (int r = 0; r < ippl::Comm->size(); ++r) {
436 if (r == Comm->rank()) {
437 std::cout << "rank: " << r << "\n";
438 std::cout << "min: " << first << "\n";
439 std::cout << "max: " << last << "\n";
440 for (int i = 0; i < elementIndices.extent(0); ++i) {
441 std::cout << i << ": " << elementIndices(i) << "\n";
442 }
443 }
444 Comm->barrier();
445 }
