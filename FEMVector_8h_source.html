413 template <typename T>
414 T innerProduct(const FEMVector<T>& a, const FEMVector<T>& b) {
415 T localSum = 0;
416 auto aView = a.getView();
417 auto bView = b.getView();
418
419
420 size_t n = aView.extent(0);
421 Kokkos::parallel_reduce("FEMVector innerProduct", n,
422 KOKKOS_LAMBDA(const size_t i, T& val){
423 val += aView(i)*bView(i);
424 },
425 Kokkos::Sum<T>(localSum)
426 );
427
428 T globalSum = 0;
429 ippl::Comm->allreduce(localSum, globalSum, 1, std::plus<T>());
430 return globalSum;
431 }

433 template <typename T>
434 T norm(const FEMVector<T>& v, int p = 2) {
435
436 T local = 0;
437 auto view = v.getView();
438 size_t n = view.extent(0);
439 switch (p) {
440 case 0: {
441 Kokkos::parallel_reduce("FEMVector l0 norm", n,
442 KOKKOS_LAMBDA(const size_t i, T& val) {
443 val = Kokkos::max(val, Kokkos::abs(view(i)));
444 },
445 Kokkos::Max<T>(local)
446 );
447 T globalMax = 0;
448 ippl::Comm->allreduce(local, globalMax, 1, std::greater<T>());
449 return globalMax;
450 }
451 default: {
452 Kokkos::parallel_reduce("FEMVector lp norm", n,
453 KOKKOS_LAMBDA(const size_t i, T& val) {
454 val += std::pow(Kokkos::abs(view(i)), p);
455 },
456 Kokkos::Sum<T>(local)
457 );
458 T globalSum = 0;
459 ippl::Comm->allreduce(local, globalSum, 1, std::plus<T>());
460 return std::pow(globalSum, 1.0 / p);
461 }
462 }
