49 template <typename T, unsigned Dim, unsigned Order, typename ElementType,
50 typename QuadratureType, typename FieldLHS, typename FieldRHS>
51 void LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS,
52 FieldRHS>::initializeElementIndices(const Layout_t& layout) {
53 layout_m = layout;
54 const auto& ldom = layout.getLocalNDIndex();
55 int npoints = ldom.size();
56 auto first = ldom.first();
57 auto last = ldom.last();
58 ippl::Vector<double, Dim> bounds;
59
60 for (size_t d = 0; d < Dim; ++d) {
61 bounds[d] = this->nr_m[d] - 1;
62 }
63
64 int upperBoundaryPoints = -1;
65
66 Kokkos::View<size_t*> points("ComputeMapping", npoints);
67 Kokkos::parallel_reduce(
68 "ComputePoints", npoints,
69 KOKKOS_CLASS_LAMBDA(const int i, int& local) {
70 int idx = i;
71 indices_t val;
72 bool isBoundary = false;
73 for (unsigned int d = 0; d < Dim; ++d) {
74 int range = last[d] - first[d] + 1;
75 val[d] = first[d] + (idx % range);
76 idx /= range;
77 if (val[d] == bounds[d]) {
78 isBoundary = true;
79 }
80 }
81 points(i) = (!isBoundary) * (this->getElementIndex(val));
82 local += isBoundary;
83 },
84 Kokkos::Sum<int>(upperBoundaryPoints));
85 Kokkos::fence();
86
87 int elementsPerRank = npoints - upperBoundaryPoints;
88 elementIndices = Kokkos::View<size_t*>("i", elementsPerRank);
89 Kokkos::View<size_t> index("index");
90
91 Kokkos::parallel_for(
92 "RemoveNaNs", npoints, KOKKOS_CLASS_LAMBDA(const int i) {
93 if ((points(i) != 0) || (i == 0)) {
94 const size_t idx = Kokkos::atomic_fetch_add(&index(), 1);
95 elementIndices(idx) = points(i);
96 }
97 });
98 }

178 KOKKOS_FUNCTION Vector<size_t, LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType,
179 FieldLHS, FieldRHS>::numElementDOFs>
180 LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS,
181 FieldRHS>::getGlobalDOFIndices(const size_t& elementIndex) const {
182 Vector<size_t, this->numElementDOFs> globalDOFs(0);
183
184 // get element pos
185 indices_t elementPos = this->getElementNDIndex(elementIndex);
186
187 // Compute the vector to multiply the ndindex with
188 ippl::Vector<size_t, Dim> vec(1);
189 for (size_t d = 1; d < dim; ++d) {
190 for (size_t d2 = d; d2 < Dim; ++d2) {
191 vec[d2] *= this->nr_m[d - 1];
192 }
193 }
194 vec *= Order; // Multiply each dimension by the order
195 size_t smallestGlobalDOF = elementPos.dot(vec);
196
197 // Add the vertex DOFs
198 globalDOFs[0] = smallestGlobalDOF;
199 globalDOFs[1] = smallestGlobalDOF + Order;
200
201 if (Dim >= 2) {
202 globalDOFs[2] = globalDOFs[1] + this->nr_m[1] * Order;
203 globalDOFs[3] = globalDOFs[0] + this->nr_m[1] * Order;
204 }
205 if (Dim >= 3) {
206 globalDOFs[4] = globalDOFs[0] + this->nr_m[1] * this->nr_m[2] * Order;
207 globalDOFs[5] = globalDOFs[1] + this->nr_m[1] * this->nr_m[2] * Order;
208 globalDOFs[6] = globalDOFs[2] + this->nr_m[1] * this->nr_m[2] * Order;
209 globalDOFs[7] = globalDOFs[3] + this->nr_m[1] * this->nr_m[2] * Order;
210 }
211
212 if (Order > 1) {
213 // If the order is greater than 1, there are edge and face DOFs, otherwise the work is
214 // done
215
216 // Add the edge DOFs
217 if (Dim >= 2) {
218 for (size_t i = 0; i < Order - 1; ++i) {
219 globalDOFs[8 + i] = globalDOFs[0] + i + 1;
220 globalDOFs[8 + Order - 1 + i] = globalDOFs[1] + (i + 1) * this->nr_m[1];
221 globalDOFs[8 + 2 * (Order - 1) + i] = globalDOFs[2] - (i + 1);
222 globalDOFs[8 + 3 * (Order - 1) + i] = globalDOFs[3] - (i + 1) * this->nr_m[1];
223 }
224 }
225 if (Dim >= 3) {
226 // TODO
227 }
228
229 // Add the face DOFs
230 if (Dim >= 2) {
231 for (size_t i = 0; i < Order - 1; ++i) {
232 for (size_t j = 0; j < Order - 1; ++j) {
233 // TODO CHECK
234 globalDOFs[8 + 4 * (Order - 1) + i * (Order - 1) + j] =
235 globalDOFs[0] + (i + 1) + (j + 1) * this->nr_m[1];
236 globalDOFs[8 + 4 * (Order - 1) + (Order - 1) * (Order - 1) + i * (Order - 1)
237 + j] = globalDOFs[1] + (i + 1) + (j + 1) * this->nr_m[1];
238 globalDOFs[8 + 4 * (Order - 1) + 2 * (Order - 1) * (Order - 1)
239 + i * (Order - 1) + j] =
240 globalDOFs[2] - (i + 1) + (j + 1) * this->nr_m[1];
241 globalDOFs[8 + 4 * (Order - 1) + 3 * (Order - 1) * (Order - 1)
242 + i * (Order - 1) + j] =
243 globalDOFs[3] - (i + 1) + (j + 1) * this->nr_m[1];
244 }
245 }
246 }
247 }
248
249 return globalDOFs;
250 }

258 KOKKOS_FUNCTION T
259 LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS, FieldRHS>::
260 evaluateRefElementShapeFunction(
261 const size_t& localDOF,
262 const LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS,
263 FieldRHS>::point_t& localPoint) const {
264 static_assert(Order == 1, "Only order 1 is supported at the moment");
265 // Assert that the local vertex index is valid.
266 assert(localDOF < this->numElementDOFs
267 && "The local vertex index is invalid"); // TODO assumes 1st order Lagrange
268
269 assert(this->ref_element_m.isPointInRefElement(localPoint)
270 && "Point is not in reference element");
271
272 // Get the local vertex indices for the local vertex index.
273 // TODO fix not order independent, only works for order 1
274 const point_t ref_element_point = this->ref_element_m.getLocalVertices()[localDOF];
275
276 // The variable that accumulates the product of the shape functions.
277 T product = 1;
278
279 for (size_t d = 0; d < Dim; d++) {
280 if (localPoint[d] < ref_element_point[d]) {
281 product *= localPoint[d];
282 } else {
283 product *= 1.0 - localPoint[d];
284 }
285 }
286
287 return product;
288 }

290 template <typename T, unsigned Dim, unsigned Order, typename ElementType,
291 typename QuadratureType, typename FieldLHS, typename FieldRHS>
292 KOKKOS_FUNCTION typename LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS,
293 FieldRHS>::point_t
294 LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS, FieldRHS>::
295 evaluateRefElementShapeFunctionGradient(
296 const size_t& localDOF,
297 const LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS,
298 FieldRHS>::point_t& localPoint) const {
299 // TODO fix not order independent, only works for order 1
300 static_assert(Order == 1 && "Only order 1 is supported at the moment");
301
302 // Assert that the local vertex index is valid.
303 assert(localDOF < this->numElementDOFs && "The local vertex index is invalid");
304
305 assert(this->ref_element_m.isPointInRefElement(localPoint)
306 && "Point is not in reference element");
307
308 // Get the local dof nd_index
309 const vertex_points_t local_vertex_points = this->ref_element_m.getLocalVertices();
310
311 const point_t& local_vertex_point = local_vertex_points[localDOF];
312
313 point_t gradient(1);
314
315 // To construct the gradient we need to loop over the dimensions and multiply the
316 // shape functions in each dimension except the current one. The one of the current
317 // dimension is replaced by the derivative of the shape function in that dimension,
318 // which is either 1 or -1.
319 for (size_t d = 0; d < Dim; d++) {
320 // The variable that accumulates the product of the shape functions.
321 T product = 1;
322
323 for (size_t d2 = 0; d2 < Dim; d2++) {
324 if (d2 == d) {
325 if (localPoint[d] < local_vertex_point[d]) {
326 product *= 1;
327 } else {
328 product *= -1;
329 }
330 } else {
331 if (localPoint[d2] < local_vertex_point[d2]) {
332 product *= localPoint[d2];
333 } else {
334 product *= 1.0 - localPoint[d2];
335 }
336 }
337 }
338
339 gradient[d] = product;
340 }
341
342 return gradient;
343 }

349 template <typename T, unsigned Dim, unsigned Order, typename ElementType,
350 typename QuadratureType, typename FieldLHS, typename FieldRHS>
351 template <typename F>
352 FieldLHS LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS,
353 FieldRHS>::evaluateAx(FieldLHS& vector, F& evalFunction) const {
354 Inform m("");
355
356 // start a timer
357 static IpplTimings::TimerRef evalAx = IpplTimings::getTimer("evaluateAx");
358 IpplTimings::startTimer(evalAx);
359
360
361 // create a new field for result with view initialized to zero (views are initialized to
362 // zero by default)
363 FieldLHS resultVector = vector.deepCopy();
364 resultVector = 0;
365
366 // List of quadrature weights
367 const Vector<T, QuadratureType::numElementNodes> w =
368 this->quadrature_m.getWeightsForRefElement();
369
370 // List of quadrature nodes
371 const Vector<point_t, QuadratureType::numElementNodes> q =
372 this->quadrature_m.getIntegrationNodesForRefElement();
373
374 // TODO move outside of evaluateAx (I think it is possible for other problems as well)
375 // Gradients of the basis functions for the DOF at the quadrature nodes
376 Vector<Vector<point_t, this->numElementDOFs>, QuadratureType::numElementNodes> grad_b_q;
377 for (size_t k = 0; k < QuadratureType::numElementNodes; ++k) {
378 for (size_t i = 0; i < this->numElementDOFs; ++i) {
379 grad_b_q[k][i] = this->evaluateRefElementShapeFunctionGradient(i, q[k]);
380 }
381 }
382
383 // Get field data and atomic result data,
384 // since it will be added to during the kokkos loop
385 ViewType view = vector.getView();
386 AtomicViewType resultView = resultVector.getView();
387
388
389 // Get domain information
390 auto ldom = layout_m.getLocalNDIndex();
391
392 using exec_space = typename Kokkos::View<const size_t*>::execution_space;
393 using policy_type = Kokkos::RangePolicy<exec_space>;
394
395 // start a timer
396 static IpplTimings::TimerRef outer_loop = IpplTimings::getTimer("evaluateAx: outer loop");
397 IpplTimings::startTimer(outer_loop);
398
399
400 // information to translate the ndindices to FEMVector positions.
401 // We add the plus one, because we have that last does not correspond
402 // to the position end+1, but simply to end
403 ippl::Vector<size_t, Dim> extents = (ldom.last()+1) - ldom.first() + 2;
404 ippl::Vector<size_t, Dim> v(1);
405 for (size_t i = 1; i < Dim; ++i) {
406 for(size_t j = i; j < Dim; ++j) {
407 v(j) *= extents(i-1);
408 }
409 }
410
411 // Loop over elements to compute contributions
412 Kokkos::parallel_for(
413 "Loop over elements", policy_type(0, elementIndices.extent(0)),
414 KOKKOS_CLASS_LAMBDA(const size_t index) {
415 const size_t elementIndex = elementIndices(index);
416 const Vector<size_t, this->numElementDOFs> local_dof = this->getLocalDOFIndices();
417 const Vector<size_t, this->numElementDOFs> global_dofs =
418 this->getGlobalDOFIndices(elementIndex);
419 Vector<indices_t, this->numElementDOFs> global_dof_ndindices;
420
421 for (size_t i = 0; i < this->numElementDOFs; ++i) {
422 global_dof_ndindices[i] = this->getMeshVertexNDIndex(global_dofs[i]);
423 }
424
425 // local DOF indices
426 size_t i, j;
427
428 // Element matrix
429 Vector<Vector<T, this->numElementDOFs>, this->numElementDOFs> A_K;
430
431 // 1. Compute the Galerkin element matrix A_K
432 for (i = 0; i < this->numElementDOFs; ++i) {
433 for (j = 0; j < this->numElementDOFs; ++j) {
434 A_K[i][j] = 0.0;
435 for (size_t k = 0; k < QuadratureType::numElementNodes; ++k) {
436 A_K[i][j] += w[k] * evalFunction(i, j, grad_b_q[k]);
437 }
438 }
439 }
440
441 // global DOF n-dimensional indices (Vector of N indices representing indices in
442 // each dimension)
443 indices_t I_nd, J_nd;
444 size_t I, J;
445 // 2. Compute the contribution to resultAx = A*x with A_K
446 for (i = 0; i < this->numElementDOFs; ++i) {
447 I_nd = global_dof_ndindices[i];
448
449 // Skip boundary DOFs (Zero Dirichlet BCs)
450 if (this->isDOFOnBoundary(I_nd)) {
451 continue;
452 }
453
454 // get the appropriate index for the Kokkos view of the field
455 I = 0;
456 for (unsigned d = 0; d < Dim; ++d) {
457 I_nd[d] = I_nd[d] - ldom[d].first() + 1;
458 I += v[d]*I_nd[d];
459 }
460
461 for (j = 0; j < this->numElementDOFs; ++j) {
462 J_nd = global_dof_ndindices[j];
463
464 // Skip boundary DOFs (Zero Dirichlet BCs)
465 if (this->isDOFOnBoundary(J_nd)) {
466 continue;
467 }
468
469 // get the appropriate index for the Kokkos view of the field
470 J = 0;
471 for (unsigned d = 0; d < Dim; ++d) {
472 J_nd[d] = J_nd[d] - ldom[d].first() + 1;
473 J += v[d]*J_nd[d];
474 }
475
476 resultView(I) += A_K[i][j] * view(J);
477 }
478 }
479 });
480 IpplTimings::stopTimer(outer_loop);
481
482 IpplTimings::stopTimer(evalAx);
483
484 return resultVector;
485 }

487 template <typename T, unsigned Dim, unsigned Order, typename ElementType,
488 typename QuadratureType, typename FieldLHS, typename FieldRHS>
489 void LagrangeSpaceFEMVector<T, Dim, Order, ElementType, QuadratureType, FieldLHS,
490 FieldRHS>::evaluateLoadVector(FieldLHS& vector) const {
491 Inform m("");
492
493 // start a timer
494 static IpplTimings::TimerRef evalLoadV = IpplTimings::getTimer("evaluateLoadVector");
495 IpplTimings::startTimer(evalLoadV);
496
497 // List of quadrature weights
498 const Vector<T, QuadratureType::numElementNodes> w =
499 this->quadrature_m.getWeightsForRefElement();
500
501 // List of quadrature nodes
502 const Vector<point_t, QuadratureType::numElementNodes> q =
503 this->quadrature_m.getIntegrationNodesForRefElement();
504
505 const indices_t zeroNdIndex = Vector<size_t, Dim>(0);
506
507 // Evaluate the basis functions for the DOF at the quadrature nodes
508 Vector<Vector<T, this->numElementDOFs>, QuadratureType::numElementNodes> basis_q;
509 for (size_t k = 0; k < QuadratureType::numElementNodes; ++k) {
510 for (size_t i = 0; i < this->numElementDOFs; ++i) {
511 basis_q[k][i] = this->evaluateRefElementShapeFunction(i, q[k]);
512 }
513 }
514
515 // Absolute value of det Phi_K
516 const T absDetDPhi = Kokkos::abs(this->ref_element_m.getDeterminantOfTransformationJacobian(
517 this->getElementMeshVertexPoints(zeroNdIndex)));
518
519 // Get domain information and ghost cells
520 auto ldom = layout_m.getLocalNDIndex();
521
522
523 // Get boundary conditions from field
524 FieldLHS temp_vector = vector.deepCopy();
525 temp_vector = 0;
526
527 // Get field data and make it atomic,
528 // since it will be added to during the kokkos loop
529 // We work with a temporary field since we need to use field
530 // to evaluate the load vector; then we assign temp to RHS field
531 AtomicViewType atomic_view = temp_vector.getView();
532 ViewType view = vector.getView();
533
534 using exec_space = typename Kokkos::View<const size_t*>::execution_space;
535 using policy_type = Kokkos::RangePolicy<exec_space>;
536
537 // start a timer
538 static IpplTimings::TimerRef outer_loop =
539 IpplTimings::getTimer("evaluateLoadVec: outer loop");
540 IpplTimings::startTimer(outer_loop);
541
542 // information to translate the ndindices to FEMVector positions.
543 // We add the plus one, because we have that last does not correspond
544 // to the position end+1, but simply to end
545 ippl::Vector<size_t, Dim> extents = (ldom.last()+1) - ldom.first() + 2;
546 ippl::Vector<size_t, Dim> v(1);
547 for (size_t i = 1; i < Dim; ++i) {
548 for(size_t j = i; j < Dim; ++j) {
549 v(j) *= extents(i-1);
550 }
551 }
552 // Loop over elements to compute contributions
553 Kokkos::parallel_for(
554 "Loop over elements", policy_type(0, elementIndices.extent(0)),
555 KOKKOS_CLASS_LAMBDA(size_t index) {
556 const size_t elementIndex = elementIndices(index);
557 const Vector<size_t, this->numElementDOFs> local_dofs = this->getLocalDOFIndices();
558 const Vector<size_t, this->numElementDOFs> global_dofs =
559 this->getGlobalDOFIndices(elementIndex);
560
561 size_t i, I;
562 size_t II, JJ;
563
564 // 1. Compute b_K
565 for (i = 0; i < this->numElementDOFs; ++i) {
566 I = global_dofs[i];
567
568 // TODO fix for higher order
569 auto dof_ndindex_I = this->getMeshVertexNDIndex(I);
570
571 if (this->isDOFOnBoundary(dof_ndindex_I)) {
572 continue;
573 }
574
575 // calculate the contribution of this element
576 T contrib = 0;
577 for (size_t k = 0; k < QuadratureType::numElementNodes; ++k) {
578 T val = 0;
579 for (size_t j = 0; j < this->numElementDOFs; ++j) {
580 // get field index corresponding to this DOF
581 size_t J = global_dofs[j];
582 auto dof_ndindex_J = this->getMeshVertexNDIndex(J);
583 JJ = 0;
584
