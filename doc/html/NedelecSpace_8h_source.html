378  KOKKOS_FUNCTION bool isDOFOnBoundary(const size_t& dofIdx) const {
379  assert(Dim == 2 && "AHHHH Not implemented");
380  size_t nx = this->nr_m[0];
381  size_t ny = this->nr_m[1];
382 
383  bool onBoundary = false;
384 
385  // South
386  bool sVal = (dofIdx < nx -1);
387  onBoundary = onBoundary || sVal;
388  // North
389  onBoundary = onBoundary || (dofIdx > nx*(ny-1) + ny*(nx-1) - nx);
390  // West
391  onBoundary = onBoundary || ((dofIdx - (nx-1)) % (2*nx - 1) == 0);
392  // East
393  onBoundary = onBoundary || ((dofIdx - 2*nx + 2) % (2*nx - 1) == 0);
394 
395  return onBoundary;
396  }
